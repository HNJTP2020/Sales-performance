<script>

let staff = [

{
name:"Queenie",
budget:150000,
sales:54749
},

{
name:"Faried",
budget:176318,
sales:48570
},

{
name:"Liyana",
budget:160000,
sales:46419
},

{
name:"Aniq",
budget:160000,
sales:23456
},

{
name:"Syahrul",
budget:80000,
sales:18614
}

];

let chart;

function updateDashboard(){

staff.sort((a,b)=>b.sales-a.sales);

let podiumHTML = `
<div class="rank-box silver">
<h2>🥈 ${staff[1].name}</h2>
<h3>RM${staff[1].sales.toLocaleString()}</h3>
</div>

<div class="rank-box gold">
<h2>👑 ${staff[0].name}</h2>
<h3>RM${staff[0].sales.toLocaleString()}</h3>
</div>

<div class="rank-box bronze">
<h2>🥉 ${staff[2].name}</h2>
<h3>RM${staff[2].sales.toLocaleString()}</h3>
</div>
`;

document.getElementById("podium").innerHTML = podiumHTML;

let totalSales = staff.reduce((a,b)=>a+b.sales,0);
let totalBudget = staff.reduce((a,b)=>a+b.budget,0);

document.getElementById("totalSales").innerHTML =
"RM"+totalSales.toLocaleString();

document.getElementById("budgetTotal").innerHTML =
"RM"+totalBudget.toLocaleString();

document.getElementById("achievement").innerHTML =
((totalSales/totalBudget)*100).toFixed(1)+"%";

document.getElementById("todayChampion").innerHTML =
`🔥 Champion : ${staff[0].name} | RM${staff[0].sales.toLocaleString()}`;

drawChart();
}

function drawChart(){

let labels = staff.map(x=>x.name);
let values = staff.map(x=>x.sales);

let colors = [
"#FFD700",
"#C0C0C0",
"#CD7F32",
"#3B82F6",
"#22C55E"
];

if(chart){
chart.destroy();
}

chart = new Chart(
document.getElementById("salesChart"),
{
type:"bar",
data:{
labels:labels,
datasets:[
{
label:"MTD Sales",
data:values,
backgroundColor:colors
}
]
},
options:{
responsive:true,
plugins:{
legend:{
display:false
}
}
}
}
);
}

function addSales(){

staff[0].sales += Number(document.getElementById("q").value || 0);
staff[1].sales += Number(document.getElementById("f").value || 0);
staff[2].sales += Number(document.getElementById("l").value || 0);
staff[3].sales += Number(document.getElementById("a").value || 0);
staff[4].sales += Number(document.getElementById("s").value || 0);

document.querySelectorAll("input").forEach(i=>i.value="");

updateDashboard();

}

updateDashboard();
